% \morandicontents prints a contents page with the current section picked out.
% Call it where a long talk needs one; do not put one before every section.
\morandicontents

\section{A first section}

\begin{frame}{A page of prose}
  \textbf{The lead states the point} and the sentences after it carry it.
  Nothing else on the page competes with it.

  \medskip
  A second paragraph, to show where a line wraps and how the rag is limited:
  a line may end up to 9mm early and no further, which keeps the right edge
  from looking torn.
\end{frame}

\begin{frame}{Points and a block}
  \begin{itemize}
    \item The first point, stated in a sentence.
    \item The second, longer, to show where a bullet wraps and how the second
          line sits under the first.
  \end{itemize}

  \medskip
  \begin{block}{A named block}
    The block label is set in the display face on the one mid tone in this
    palette that can carry white type.
  \end{block}
\end{frame}
