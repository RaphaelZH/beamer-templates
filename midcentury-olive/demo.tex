\tdsection{Cards, tables and code}{%
  The three constructs that carry structure, each at the width it was measured
  for.}

\begin{frame}{Three cards}

  \begin{stepflow}{3}
    \stepcard{a source}{%
      what goes in, described in the two or three lines a card holds before it
      starts setting the height of the row}
    \steparrow
    \stepcard{a process}{%
      what happens to it — \ac{a marked term}, and a \code{literal} that
      follows the card's own size}
    \steparrow
    \stepcard{a result}{%
      what comes out, and the one fact about it worth carrying to the next
      page}
  \end{stepflow}

  \vspace{0.7em}
  {\small
  \acb{The prose under a card row says what the cards cannot.} Restating a card
  in the paragraph beneath it is the commonest way one of these pages runs
  over. \lipsum[5][1-2]\par}

\end{frame}

\begin{frame}{Four cards, and a width multiplier}

  \begin{stepflow}{4}
    \stepcard[0.8]{short}{a card whose body is brief}
    \steparrow
    \stepcard[1.2]{longer}{%
      a card carrying more, given a wider box so the row does not have to be as
      tall as this one alone would make it}
    \steparrow
    \stepcard[1.2]{longer}{%
      the multipliers across a row should add up to the number of cards — here
      0.8 + 1.2 + 1.2 + 0.8}
    \steparrow
    \stepcard[0.8]{short}{brief again}
  \end{stepflow}

  \vspace{0.7em}
  {\small Widening a card does not shorten the row: its height is set by
  whichever card runs to the most lines. Use the multiplier to even out the
  line counts, not to buy space.\par}

\end{frame}

\begin{frame}{A table}

  {\footnotesize
  \begin{tabular}{@{}>{\raggedright\arraybackslash}p{3.0cm}
                    >{\raggedright\arraybackslash}p{4.8cm}
                    >{\raggedright\arraybackslash}p{5.0cm}@{}}
    \toprule
    & \thead{first heading} & \thead{second heading} \\
    \midrule
    \textbf{short label} & a cell of the length these columns were measured
      for, which is two lines & \ac{a marked phrase}, and then the rest \\
    \addlinespace[0.4em]
    \textbf{short label} & another two-line cell, so the row heights are even
      down the table & \code{a literal}, upright and at the table's own size \\
    \addlinespace[0.4em]
    \textbf{short label} & keep the label short: a wrapped label sets the row
      height even when both cells beside it are one line & the third column is
      widest because it usually carries the consequence \\
    \bottomrule
  \end{tabular}\par}

  \vspace{0.6em}
  {\small A sentence under a table says what the reader should take from it.
  Without one, a table is a list of facts with no claim attached.\par}

\end{frame}

\begin{frame}[fragile]{Code beside prose}

  \begin{columns}[T]

      Two things worth pointing at: \code{nested} and \code{note}.\par}
    \end{column}
  \end{columns}

\end{frame}

  ]
    \node[key] (a) at (0,0)     {\code{start}};
    \node[nd]  (b) at (3.0,0.7) {\code{one way}};
    \node[nd]  (c) at (3.0,-0.7){\code{another}};

    \draw[ed] (b) -- (d);
    \draw[alt] (a) -- (c) node[lab, midway, below, sloped, text=oliveRust]

  \vspace{0.5em}
  {\raggedright\fontsize{6.6}{8.4}\selectfont\color{oliveInk!70}%
   A note under a figure, at the figure's own size — apparatus should never be
   larger than what it annotates. \textcolor{oliveRust}{The alert colour} marks
   the one path that is different in kind, not merely different.\par}

  \vspace{0.5em}
  {\raggedright\fontsize{6.6}{8.4}\selectfont\color{oliveInk!70}%
   \setlength{\parskip}{0.2em}%
   \srcline{\textbf{A bulleted note}, hanging so the second line clears the
   bullet. Defined in the preamble — a command with an argument does not
   survive beamer's second read of a frame body.}
   \srcline{\textbf{A second}, to show the spacing between them.}}

\end{frame}

\begin{frame}{What the build tells you}

  \slidelead{Nothing here scales to fit. If a page holds more than the frame
  does, \code{./check.sh} names the frame and stops.}

  {\small\setlength{\parskip}{0.6em}%
  \acb{That is the whole design.} A deck whose type size changes from page to
  page is worse than a deck that is one page longer, and auto-scaling produces
  the first without ever saying so. \lipsum[6][1-2]

  \acb{The log is filtered, not ignored.} One warning per titled frame is the
  full-bleed title band and is suppressed by count; anything else is real.
  \code{README.md} lists what was verified and why.\par}

  \aside{Try it: add a paragraph to any page above and rebuild. The frame will
  be named, with the amount it runs over in points.}

\end{frame}
